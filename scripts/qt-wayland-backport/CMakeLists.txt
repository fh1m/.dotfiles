cmake_minimum_required(VERSION 3.22)
project(SenseiWaylandBackport VERSION 6.12.0 LANGUAGES C CXX)
if(NOT DEFINED SENSEI_QT_SOURCE_DIR)
    message(FATAL_ERROR "Pass the pinned Qt source directory explicitly")
endif()
set(CMAKE_SKIP_RPATH ON)
set(QT_REPO_MODULE_VERSION "6.12.0")
set(QT_SKIP_AUTO_PLUGIN_INCLUSION ON)
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS BuildInternals Core Gui CorePrivate GuiPrivate OpenGL WaylandScannerTools)
qt_build_repo_begin()
list(PREPEND CMAKE_MODULE_PATH
    "${SENSEI_QT_SOURCE_DIR}/cmake/3rdparty/extra-cmake-modules/find-modules"
    "${SENSEI_QT_SOURCE_DIR}/cmake/3rdparty/extra-cmake-modules/modules")
qt_find_package(Wayland PROVIDED_TARGETS Wayland::Client Wayland::Cursor Wayland::Egl REQUIRED COMPONENTS Client Cursor Egl)
qt_find_package(WaylandScanner PROVIDED_TARGETS Wayland::Scanner REQUIRED)
qt_find_package(EGL PROVIDED_TARGETS EGL::EGL REQUIRED)
set(QT_FEATURE_waylandscanner ON)
set(QT_FEATURE_wayland ON)
add_subdirectory("${SENSEI_QT_SOURCE_DIR}/src/plugins/platforms/wayland" wayland-client)
qt_build_repo_end()
